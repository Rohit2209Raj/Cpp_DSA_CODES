class Solution {
public:
    int findContentChildren(vector<int>& g, vector<int>& s) {
        sort(g.begin(),g.end());
        sort(s.begin(),s.end());
        int count=0;
        int j=0;
        int k=-199;
        for(int i=0;i<g.size();i++){
            j=k;
            while(j<s.size()){
                if(g[i]<=s[j]){
                    count++;
                    k=j;
                    break;
                }
            }
        }
        return count;
    }
};
